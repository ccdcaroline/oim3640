# ----------------------------
# Heuristics / dictionaries
# ----------------------------
PROTEIN_KWS = [
    "chicken", "turkey", "beef", "steak", "pork", "ham", "salmon", "tuna", "cod", "shrimp",
    "egg", "eggs", "tofu", "tempeh", "beans", "lentil", "chickpea", "edamame", "black bean",
    "greek yogurt", "yogurt", "cottage", "protein", "seitan"
]

VEG_KWS = [
    "broccoli", "spinach", "kale", "salad", "greens", "carrot", "zucchini", "pepper",
    "asparagus", "brussels", "cauliflower", "green bean", "cabbage", "tomato", "cucumber",
    "mushroom", "onion", "mixed vegetables", "veggie"
]

CARB_KWS = [
    "brown rice", "rice", "quinoa", "sweet potato", "potato", "wrap", "tortilla", "pita",
    "whole wheat", "wheat", "oats", "pasta", "noodle", "bread", "roll", "barley", "couscous"
]

HEALTH_DOWN_KWS = [
    "fried", "fries", "breaded", "crispy", "alfredo", "cream", "creamy", "cheesy", "nacho",
    "buffalo", "mayo", "bacon", "sausage", "pepperoni"
]

DESSERT_KWS = [
    "cookie", "brownie", "cake", "ice cream", "donut", "cupcake", "pie", "dessert", "pudding"
]

DRESSING_KWS = [
    "vinaigrette", "olive oil", "lemon", "salsa", "tahini", "hummus", "tzatziki",
    "hot sauce", "mustard", "guacamole", "pico"
]


# ----------------------------
# Data structures
# ----------------------------
@dataclass
class MenuItem:
    name: str
    raw: str
    health_penalty: int
    tags: Set[str]


# ----------------------------
# Helpers
# ----------------------------
def normalize(s: str) -> str:
    s = s.strip().lower()
    s = re.sub(r"\s+", " ", s)
    return s

def split_menu_text(text: str) -> List[str]:
    """
    Tries to split pasted menu text into item-ish lines.
    Works with bullets, commas, new lines, station headers, etc.
    """
    if not text or not text.strip():
        return []

    # Replace bullets with newlines
    t = text.replace("•", "\n").replace("·", "\n").replace("–", "-")
    # Split on newlines first
    lines = [ln.strip() for ln in t.splitlines() if ln.strip()]

    # Further split long lines with commas/semicolons if they look like lists
    items: List[str] = []
    for ln in lines:
        if len(ln) > 80 and ("," in ln or ";" in ln):
            parts = re.split(r"[;,]", ln)
            for p in parts:
                p = p.strip()
                if p:
                    items.append(p)
        else:
            items.append(ln)

    # Remove obvious station headers / noise
    noise_patterns = [
        r"^breakfast$",
        r"^lunch$",
        r"^dinner$",
        r"^brunch$",
        r"^today",
        r"^station",
        r"^performance",
        r"^plant",
        r"^homestyle",
        r"^made to order",
    ]
    cleaned = []
    for it in items:
        n = normalize(it)
        if any(re.search(pat, n) for pat in noise_patterns):
            continue
        # drop very short fragments
        if len(n) < 3:
            continue
        cleaned.append(it.strip())

    return cleaned

def classify_item(name: str) -> MenuItem:
    n = normalize(name)
    tags: Set[str] = set()

    def has_any(kws: List[str]) -> bool:
        return any(kw in n for kw in kws)

    if has_any(PROTEIN_KWS):
        tags.add("protein")
    if has_any(VEG_KWS):
        tags.add("veg")
    if has_any(CARB_KWS):
        tags.add("carb")
    if has_any(DRESSING_KWS):
        tags.add("sauce")
    if has_any(DESSERT_KWS):
        tags.add("dessert")

    penalty = 0
    if has_any(HEALTH_DOWN_KWS):
        penalty += 2
        tags.add("less_healthy")
    if "dessert" in tags:
        penalty += 3

    # simple diet tags
    if any(x in n for x in ["tofu", "tempeh", "beans", "lentil", "chickpea", "edamame", "seitan"]):
        tags.add("plant_protein")
    if any(x in n for x in ["beef", "steak", "chicken", "turkey", "pork", "ham", "salmon", "tuna", "cod", "shrimp"]):
        tags.add("animal_protein")

    return MenuItem(name=name.strip(), raw=name, health_penalty=penalty, tags=tags)

def dedupe_fuzzy(items: List[MenuItem], threshold: int = 92) -> List[MenuItem]:
    """
    Deduplicate near-duplicates (e.g., 'Grilled Chicken Breast' vs 'Grilled Chicken').
    """
    kept: List[MenuItem] = []
    for it in items:
        n = normalize(it.name)
        is_dup = False
        for k in kept:
            if fuzz.ratio(n, normalize(k.name)) >= threshold:
                is_dup = True
                break
        if not is_dup:
            kept.append(it)
    return kept

def apply_filters(items: List[MenuItem], vegetarian: bool, vegan: bool, no_pork: bool, avoid_dairy_words: bool) -> List[MenuItem]:
    out = []
    for it in items:
        n = normalize(it.name)

        if vegetarian:
            if "animal_protein" in it.tags and not any(x in n for x in ["egg", "eggs"]):
                continue
            # also exclude obvious meat words
            if any(x in n for x in ["chicken", "turkey", "beef", "steak", "pork", "ham", "salmon", "tuna", "cod", "shrimp", "bacon", "sausage", "pepperoni"]):
                continue

        if vegan:
            # exclude all animal words + eggs + dairy words
            if any(x in n for x in ["chicken", "turkey", "beef", "steak", "pork", "ham", "salmon", "tuna", "cod", "shrimp", "egg", "eggs", "cheese", "milk", "yogurt", "butter", "cream", "whey"]):
                continue

        if no_pork and any(x in n for x in ["pork", "ham", "bacon", "sausage", "pepperoni"]):
            continue

        if avoid_dairy_words and any(x in n for x in ["cheese", "milk", "yogurt", "butter", "cream", "whey"]):
            continue

        out.append(it)
    return out

def pick_weighted(items: List[MenuItem], prefer_healthy: bool = True) -> MenuItem:
    # Lower penalty = more likely when prefer_healthy
    if not items:
        raise ValueError("No items to choose from.")
    if not prefer_healthy:
        return random.choice(items)

    weights = []
    for it in items:
        # penalty 0 -> weight 6, penalty 1 -> 4, penalty 2 -> 2, penalty 3+ -> 1
        w = max(1, 6 - 2 * it.health_penalty)
        weights.append(w)
    return random.choices(items, weights=weights, k=1)[0]

def meal_signature(parts: List[str]) -> str:
    key = " | ".join(sorted(normalize(p) for p in parts))
    return hashlib.sha256(key.encode("utf-8")).hexdigest()[:12]


# ----------------------------
# Meal generator
# ----------------------------
def generate_meals(
    menu_items: List[MenuItem],
    goal: str,
    n_meals: int,
    allow_less_healthy: bool,
    require_veg: bool,
    unique: bool = True,
) -> List[Dict[str, str]]:
    proteins = [it for it in menu_items if "protein" in it.tags or "plant_protein" in it.tags]
    vegs = [it for it in menu_items if "veg" in it.tags and "dessert" not in it.tags]
    carbs = [it for it in menu_items if "carb" in it.tags and "dessert" not in it.tags]
    sauces = [it for it in menu_items if "sauce" in it.tags and "dessert" not in it.tags]

    # If menu is sparse, allow fallback buckets
    if not vegs:
        vegs = [it for it in menu_items if "dessert" not in it.tags and it.health_penalty <= 1]
    if not carbs:
        carbs = [it for it in menu_items if "dessert" not in it.tags and it.health_penalty <= 1]

    used_sigs: Set[str] = set()
    results = []

    # Prefer healthy unless user allows otherwise
    prefer_healthy = not allow_less_healthy

    for _ in range(2000):  # safety loop for uniqueness
        if len(results) >= n_meals:
            break

        chosen = []

        # Goal logic
        if goal == "High-protein":
            if proteins:
                p = pick_weighted(proteins, prefer_healthy=True)
                chosen.append(p.name)
                # optionally double protein if menu supports
                if len(proteins) >= 3 and random.random() < 0.35:
                    p2 = pick_weighted([x for x in proteins if x.name != p.name], prefer_healthy=True)
                    chosen.append(p2.name)
        elif goal == "Plant-based":
            plant_proteins = [it for it in proteins if "plant_protein" in it.tags]
            if plant_proteins:
                chosen.append(pick_weighted(plant_proteins, prefer_healthy=True).name)
            else:
                # fallback: pick healthiest protein-like items
                chosen.append(pick_weighted(proteins or menu_items, prefer_healthy=True).name)
        else:  # Balanced
            if proteins:
                chosen.append(pick_weighted(proteins, prefer_healthy=True).name)

        # Veg component
        if require_veg and vegs:
            chosen.append(pick_weighted(vegs, prefer_healthy=True).name)
        elif vegs and random.random() < 0.7:
            chosen.append(pick_weighted(vegs, prefer_healthy=True).name)

        # Carb component (optional depending on goal)
        if carbs and (goal != "High-protein" or random.random() < 0.55):
            chosen.append(pick_weighted(carbs, prefer_healthy=True).name)

        # Sauce/topping
        if sauces and random.random() < 0.6:
            chosen.append(pick_weighted(sauces, prefer_healthy=True).name)

        # Handle less-healthy restriction
        if not allow_less_healthy:
            # reject if any part is tagged less_healthy or dessert
            bad = False
            for nm in chosen:
                it = next((x for x in menu_items if x.name == nm), None)
                if it and ("less_healthy" in it.tags or "dessert" in it.tags):
                    bad = True
                    break
            if bad:
                continue

        sig = meal_signature(chosen)
        if unique and sig in used_sigs:
            continue

        used_sigs.add(sig)

        # Build a “format” the student recognizes
        style = random.choice(["Bowl", "Plate", "Salad", "Wrap (if available)", "Power Plate"])
        title = f"{goal} {style}"

        results.append({
            "Meal": title,
            "Build": " + ".join(chosen),
            "Unique ID": sig
        })

    return results

st.set_page_config(page_title="Trim Meal Idea Generator", page_icon="🥗", layout="wide")

st.title("🥗 Trim Dining Hall Healthy Meal Idea Generator (Babson)")
st.caption(
    "Paste the Trim menu (from DineOnCampus) and generate unique healthy meal ideas. "
    "Babson points students to DineOnCampus for menus/updates. "
)

with st.expander("How to use this", expanded=True):
    st.markdown(
        """
1) Open Babson’s menus on DineOnCampus and copy the menu items you see for today.  
2) Paste them below.  
3) Pick your goal + filters, then generate meals.
        """
    )

menu_text = st.text_area(
    "Paste today's Trim menu items here (one per line works best):",
    height=220,
    placeholder="Example:\nGrilled Chicken Breast\nBrown Rice\nRoasted Broccoli\nGarden Salad\nLentil Soup\nTofu Stir-Fry\n..."
)

col1, col2, col3, col4 = st.columns(4)

with col1:
    goal = st.selectbox("Goal", ["Balanced", "High-protein", "Plant-based"])

with col2:
    n_meals = st.slider("How many ideas?", 3, 20, 8)

with col3:
    require_veg = st.checkbox("Require a veggie in every meal", value=True)

with col4:
    allow_less_healthy = st.checkbox("Allow fried/creamy items", value=False)

st.subheader("Diet filters")
c1, c2, c3, c4 = st.columns(4)
with c1:
    vegetarian = st.checkbox("Vegetarian", value=False)
with c2:
    vegan = st.checkbox("Vegan", value=False)
with c3:
    no_pork = st.checkbox("No pork", value=False)
with c4:
    avoid_dairy_words = st.checkbox("Avoid dairy words (basic)", value=False)

if vegan:
    vegetarian = True  # vegan implies vegetarian-ish filtering

if st.button("Generate meal ideas ✅", type="primary"):
    raw_items = split_menu_text(menu_text)
    if not raw_items:
        st.error("Paste at least a few menu items first.")
        st.stop()

    classified = [classify_item(x) for x in raw_items]
    classified = dedupe_fuzzy(classified)

    filtered = apply_filters(classified, vegetarian, vegan, no_pork, avoid_dairy_words)

    if not filtered:
        st.error("After filters, there are no menu items left. Try loosening filters.")
        st.stop()

    meals = generate_meals(
        menu_items=filtered,
        goal=goal,
        n_meals=n_meals,
        allow_less_healthy=allow_less_healthy,
        require_veg=require_veg,
        unique=True
    )

    if not meals:
        st.warning("Couldn’t generate meals with the current menu + settings. Try allowing less-healthy items or unchecking 'require veggie'.")
        st.stop()

    df = pd.DataFrame(meals)
    st.success(f"Generated {len(df)} unique meal ideas.")
    st.dataframe(df, use_container_width=True)

    st.download_button(
        "Download as CSV",
        data=df.to_csv(index=False).encode("utf-8"),
        file_name="trim_meal_ideas.csv",
        mime="text/csv"
    )

st.divider()
st.caption(

    "it works by using the menu text you paste in. Babson directs menu updates to DineOnCampus."
)
